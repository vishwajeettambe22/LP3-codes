class Fibonacci {

    static int count = 0;

    // Recursive method
    static int recursive(int n) {
        count++;
        if (n <= 1)
            return n;
        return recursive(n - 1) + recursive(n - 2);
    }

    // Non-recursive method
    static void nonRecursive(int n) {
        int a = 0, b = 1, c, count = 0;

        for (int i = 0; i < n; i++) {
            System.out.print(a + " ");
            c = a + b;
            a = b;
            b = c;
            count++;
        }
        System.out.println("\nStep Count: " + count);
    }

    public static void main(String[] args) {
        int n = 5;

        System.out.println("Non-Recursive:");
        nonRecursive(n);

        System.out.println("Recursive:");
        count = 0;

        for (int i = 0; i < n; i++)
            System.out.print(recursive(i) + " ");

        System.out.println("\nStep Count: " + count);
    }
}

import threading
import time

# Define matrices directly
A = [
    [1, 2, 3],
    [1, 5, 1],
    [3, 4, 5]
]

B = [
    [4, 7, 6],
    [2, 9, 9],
    [5, 1, 1]
]

# Normal Matrix Multiplication
def normal_multiply(A, B):
    rows = len(A)
    cols = len(B[0])
    common = len(B)

    C = [[0 for j in range(cols)] for i in range(rows)]

    for i in range(rows):
        for j in range(cols):
            for k in range(common):
                C[i][j] += A[i][k] * B[k][j]

    return C


# Function for each thread
def multiply_row(A, B, C, row):
    for j in range(len(B[0])):
        for k in range(len(B)):
            C[row][j] += A[row][k] * B[k][j]


# Multithreaded Matrix Multiplication
def threaded_multiply(A, B):
    rows = len(A)
    cols = len(B[0])

    C = [[0 for j in range(cols)] for i in range(rows)]

    threads = []

    for i in range(rows):
        t = threading.Thread(
            target=multiply_row,
            args=(A, B, C, i)
        )
        threads.append(t)
        t.start()

    for t in threads:
        t.join()

    return C


# Check matrix dimensions
if len(A[0]) != len(B):
    print("Matrix multiplication not possible!")

else:
    # Normal multiplication
    start = time.perf_counter()
    C1 = normal_multiply(A, B)
    normal_time = time.perf_counter() - start

    # Multithreaded multiplication
    start = time.perf_counter()
    C2 = threaded_multiply(A, B)
    thread_time = time.perf_counter() - start

    print("Matrix A:")
    for row in A:
        print(row)

    print("\nMatrix B:")
    for row in B:
        print(row)

    print("\nNormal Matrix Multiplication:")
    for row in C1:
        print(row)

    print("\nMultithreaded Matrix Multiplication:")
    for row in C2:
        print(row)

    print("\nPerformance Comparison:")
    print("Normal Time:", normal_time, "seconds")
    print("Thread Time:", thread_time, "seconds")
